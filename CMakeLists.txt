cmake_minimum_required(VERSION 3.26)
project(IRForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
# Exact release tags; source overrides are supported for offline builds.
FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
add_library(irforge_core
  src/core/Event.cpp src/core/EventBus.cpp src/core/Config.cpp src/core/Clock.cpp
  src/detection/ProcessTracker.cpp src/detection/RuleEngine.cpp
  src/detection/Correlator.cpp src/detection/RiskEngine.cpp
  src/monitoring/Diff.cpp src/output/JsonlWriter.cpp src/output/AlertManager.cpp
  src/output/Metrics.cpp src/replay/ReplayEngine.cpp)
target_include_directories(irforge_core PUBLIC include)
target_link_libraries(irforge_core PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
  target_compile_definitions(irforge_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(irforge_core PUBLIC advapi32 ws2_32)
endif()
add_executable(irforge src/main.cpp)
target_link_libraries(irforge PRIVATE irforge_core)
if(WIN32)
  target_sources(irforge PRIVATE src/monitoring/ProcessMonitor.cpp src/monitoring/NetworkMonitor.cpp src/monitoring/RegistryMonitor.cpp)
  target_link_libraries(irforge PRIVATE tdh iphlpapi ole32)
endif()
foreach(t irforge_core irforge)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(MINGW)
  target_link_options(irforge PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  FetchContent_MakeAvailable(Catch2)
  add_executable(irforge_tests tests/test_core.cpp tests/test_detection.cpp tests/test_diff.cpp)
  target_link_libraries(irforge_tests PRIVATE irforge_core Catch2::Catch2WithMain)
  target_compile_definitions(irforge_tests PRIVATE IRFORGE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MINGW)
    target_link_options(irforge_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
  include(Catch)
  catch_discover_tests(irforge_tests)
  add_test(NAME rules_validate COMMAND irforge rules validate --dir ${CMAKE_CURRENT_SOURCE_DIR}/rules)
  add_test(NAME replay_chain COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:irforge> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.cmake)
endif()
install(TARGETS irforge RUNTIME DESTINATION bin)
install(DIRECTORY config rules DESTINATION .)
