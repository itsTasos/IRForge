cmake_minimum_required(VERSION 3.26)
project(IRForge VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_TESTING "Build the separately maintained local development tests" OFF)
if(BUILD_TESTING AND NOT EXISTS "${CMAKE_SOURCE_DIR}/tests/test_storage.cpp")
  message(FATAL_ERROR "Development tests are intentionally excluded from the release repository. Configure with BUILD_TESTING=OFF, or provide the complete local test suite.")
endif()
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(IRFORGE_DEPS "${CMAKE_SOURCE_DIR}/build/dependencies" CACHE PATH "Pinned dependency restore directory")
set(IRFORGE_JSON_URL "https://github.com/nlohmann/json/archive/refs/tags/v3.11.3.zip")
set(IRFORGE_CATCH_URL "https://github.com/catchorg/Catch2/archive/refs/tags/v3.7.1.zip")
if(EXISTS "${IRFORGE_DEPS}/json.zip")
  set(IRFORGE_JSON_URL "${IRFORGE_DEPS}/json.zip")
endif()
if(EXISTS "${IRFORGE_DEPS}/catch2.zip")
  set(IRFORGE_CATCH_URL "${IRFORGE_DEPS}/catch2.zip")
endif()
FetchContent_Declare(json URL "${IRFORGE_JSON_URL}" URL_HASH SHA256=04022b05d806eb5ff73023c280b68697d12b93e1b7267a0b22a1a39ec7578069 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(Catch2 URL "${IRFORGE_CATCH_URL}" URL_HASH SHA256=7d771897398704ecb61eae534912e50c4d3ec6129c4d01c174a55c29657970d7 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
include(cmake/Dependencies.cmake)
file(READ migrations/001_initial.sql IRFORGE_SQL_001)
file(READ migrations/002_provenance_aliases.sql IRFORGE_SQL_002)
configure_file(cmake/MigrationSql.hpp.in generated/MigrationSql.hpp @ONLY)
file(READ assets/attack_dataset/enterprise-v18.0-subset.json IRFORGE_ATTACK_DATA)
configure_file(cmake/AttackData.hpp.in generated/AttackData.hpp @ONLY)
add_library(irforge_core
  src/core/Event.cpp src/core/EventBus.cpp src/core/Config.cpp src/core/Clock.cpp src/core/Schema.cpp
  src/detection/ProcessTracker.cpp src/detection/RuleEngine.cpp src/detection/AttackMapper.cpp src/detection/SigmaImporter.cpp src/detection/YaraScanner.cpp src/detection/BaselineEngine.cpp
  src/detection/Correlator.cpp src/detection/RiskEngine.cpp src/detection/IncidentGraph.cpp src/remote/RemoteProtocol.cpp
  src/monitoring/Diff.cpp src/output/JsonlWriter.cpp src/output/AlertManager.cpp
  src/output/Metrics.cpp src/replay/ReplayEngine.cpp src/storage/SqliteStore.cpp
  src/enrichment/HashEngine.cpp src/enrichment/ArtifactEnricher.cpp
  src/monitoring/FileMonitor.cpp src/monitoring/InventoryMonitor.cpp src/windows/EventLogSubscriber.cpp src/ipc/PipeServer.cpp src/reporting/Evidence.cpp src/reporting/ReportExporter.cpp src/ti/VirusTotal.cpp src/ti/TiEnricher.cpp src/cli/Commands.cpp)
target_include_directories(irforge_core PUBLIC include PRIVATE "${CMAKE_BINARY_DIR}/generated")
set(IRFORGE_YARA_SDK "${CMAKE_SOURCE_DIR}/build/dependencies/yara" CACHE PATH "Pinned YARA-X C API SDK")
if(NOT EXISTS "${IRFORGE_YARA_SDK}/yara_x.h" OR NOT EXISTS "${IRFORGE_YARA_SDK}/yara_x_capi.dll")
  message(FATAL_ERROR "Pinned YARA-X 1.21.0 SDK absent: restore dependencies before configuring")
endif()
target_include_directories(irforge_core PRIVATE "${IRFORGE_YARA_SDK}")
target_link_libraries(irforge_core PUBLIC nlohmann_json::nlohmann_json irforge_sqlite tinyxml2::tinyxml2 yaml-cpp::yaml-cpp)
if(WIN32)
  target_compile_definitions(irforge_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_sources(irforge_core PRIVATE src/monitoring/ProcessMonitor.cpp src/monitoring/NetworkMonitor.cpp src/monitoring/RegistryMonitor.cpp src/service/WindowsService.cpp)
  target_link_libraries(irforge_core PUBLIC advapi32 ws2_32 ole32 oleaut32 bcrypt wintrust crypt32 shell32 uuid taskschd iphlpapi wevtapi winhttp tdh)
endif()
add_executable(irforge src/main.cpp)
target_link_libraries(irforge PRIVATE irforge_core)
add_custom_command(TARGET irforge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${IRFORGE_YARA_SDK}/yara_x_capi.dll" "$<TARGET_FILE_DIR:irforge>/yara_x_capi.dll")
set(Qt6_DIR "${CMAKE_SOURCE_DIR}/build/dependencies/qt/lib/cmake/Qt6" CACHE PATH "Pinned Qt 6.8.3 configuration")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Gui PrintSupport Concurrent Network)
add_library(irforge_remote src/remote/TlsTransport.cpp)
target_link_libraries(irforge_remote PUBLIC irforge_core Qt6::Core Qt6::Network)
target_link_libraries(irforge_remote PRIVATE cabinet)
target_link_libraries(irforge PRIVATE irforge_remote)
add_executable(irforge_collector src/remote/collector_main.cpp)
target_link_libraries(irforge_collector PRIVATE irforge_remote)
add_executable(irforge_gui WIN32 src/ui/main.cpp src/ui/MainWindow.cpp)
target_include_directories(irforge_gui PRIVATE src/ui)
target_link_libraries(irforge_gui PRIVATE irforge_core Qt6::Widgets Qt6::Gui Qt6::PrintSupport Qt6::Concurrent)
if(MINGW)
  target_link_options(irforge_gui PRIVATE -static-libgcc -static-libstdc++)
endif()
if(WIN32)
  target_link_libraries(irforge PRIVATE tdh iphlpapi ole32)
endif()
foreach(t irforge_core irforge irforge_remote irforge_collector irforge_gui)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(MINGW)
  target_link_options(irforge PRIVATE -municode -static-libgcc -static-libstdc++)
  target_link_options(irforge_collector PRIVATE -static-libgcc -static-libstdc++)
endif()
include(CTest)
option(IRFORGE_COVERAGE "Instrument owned GNU C++ targets for gcov (development builds only)" OFF)
if(IRFORGE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT BUILD_TESTING)
    message(FATAL_ERROR "IRFORGE_COVERAGE requires GNU and BUILD_TESTING=ON")
  endif()
  foreach(t irforge_core irforge irforge_remote irforge_collector irforge_gui)
    target_compile_options(${t} PRIVATE --coverage -O0 -g)
    target_link_options(${t} PUBLIC --coverage)
  endforeach()
endif()
if(BUILD_TESTING)
  FetchContent_MakeAvailable(Catch2)
  add_executable(irforge_tests tests/test_core.cpp tests/test_detection.cpp tests/test_diff.cpp tests/test_storage.cpp tests/test_native_expansion.cpp tests/test_provider_rules.cpp tests/test_analysis.cpp tests/test_ipc_reports.cpp tests/test_ti.cpp tests/test_service.cpp tests/test_remote.cpp tests/test_hardening.cpp tests/test_context.cpp)
  add_executable(irforge_yara_tests tests/yara_lifecycle.cpp)
  target_link_libraries(irforge_yara_tests PRIVATE irforge_core)
  if(MINGW)
    target_link_options(irforge_yara_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME yara_cancel_reload_watchdog COMMAND irforge_yara_tests)
  add_executable(irforge_crash_writer tests/crash_writer.cpp)
  add_executable(irforge_parser_fuzz tests/fuzz/parsers.cpp)
  target_link_libraries(irforge_parser_fuzz PRIVATE irforge_core)
  if(MINGW)
    target_link_options(irforge_parser_fuzz PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME parser_fuzz_seeded COMMAND irforge_parser_fuzz)
  set_tests_properties(parser_fuzz_seeded PROPERTIES TIMEOUT 60)
  target_link_libraries(irforge_crash_writer PRIVATE irforge_core)
  if(MINGW)
    target_link_options(irforge_crash_writer PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  target_link_libraries(irforge_tests PRIVATE irforge_core Catch2::Catch2WithMain)
  target_compile_definitions(irforge_tests PRIVATE IRFORGE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MINGW)
    target_link_options(irforge_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
  include(Catch)
  catch_discover_tests(irforge_tests)
  add_test(NAME rules_validate COMMAND irforge rules validate --dir ${CMAKE_CURRENT_SOURCE_DIR}/rules)
  add_test(NAME replay_chain COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:irforge> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.cmake)
  add_test(NAME yara_official_capi COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:irforge> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/yara.cmake)
  add_test(NAME sqlite_forced_crash_recovery COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:irforge> -DWRITER=$<TARGET_FILE:irforge_crash_writer> -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/recovery.cmake)
  set(IRFORGE_TEST_PYTHON "$ENV{USERPROFILE}/.cache/codex-runtimes/codex-primary-runtime/dependencies/python/python.exe" CACHE FILEPATH "Python with cryptography for real TLS integration fixtures")
  if(EXISTS "${IRFORGE_TEST_PYTHON}")
    add_test(NAME remote_real_mtls COMMAND "${IRFORGE_TEST_PYTHON}" "${CMAKE_SOURCE_DIR}/tests/remote_tls.py" --binary "$<TARGET_FILE_DIR:irforge>" --source "${CMAKE_SOURCE_DIR}" --output "${CMAKE_BINARY_DIR}/tls-acceptance")
    set_tests_properties(remote_real_mtls PROPERTIES TIMEOUT 60)
  endif()
endif()
install(TARGETS irforge irforge_gui irforge_collector RUNTIME DESTINATION bin)
